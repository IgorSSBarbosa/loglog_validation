\begin{abstract}
\noindent
This note derives a maximum-likelihood estimator (MLE) for the critical
exponent $\gamma$ that is \emph{not} the log-log OLS estimator already
implemented in \texttt{tools/loglog.py}. That estimator works by taking
$\log \overline{Y}_i$ as (approximately) Gaussian and fitting a line. Here we
instead take $\overline{Y}_i$ itself as (approximately) Gaussian, motivated
directly by the classical CLT applied to a sample mean, and derive the MLE for
$\gamma$ under that model. The two estimators are different objects, both
CLT-motivated, and this note also records exactly how they differ
(Remark~\ref{rmk:vs-ols}). \textbf{Revision 1:} the first version of this
note derived only first-order (stationarity) conditions and did not check
that the stationary point found is actually a maximum. Section~\ref{sec:second-order}
fixes that: it proves concavity of the likelihood at fixed $\sigma^2$, shows
by explicit counterexample that this does \emph{not} extend to the profiled
objective $\Phi$ of \eqref{eq:objective}, and derives a coordinate-ascent
algorithm that only ever solves the part that is provably well-posed.
\textbf{Revision 2:} Section~\ref{sec:direct} reconsiders whether that
complexity is warranted --- direct joint optimization over all three
parameters has exactly the same \emph{unproven} global-optimality status as
coordinate ascent (neither the joint likelihood nor its profile is
established to be globally concave), so the extra machinery buys a
structural guarantee that turns out not to be the thing doing the actual
work; the same post-hoc diagnostics are required either way. Section~\ref{sec:implementation}
now describes the direct approach, \textbf{implemented in
\texttt{tools/loglog.py}'s \texttt{gamma\_mle}} and validated as described
there.
\end{abstract}

\section{Setup and assumptions}

We are given data at $K$ distinct scales $i_1 < i_2 < \cdots < i_K$. At scale
$i_k$, we have $n_k$ i.i.d.\ samples $Y_k^{(1)}, \dots, Y_k^{(n_k)}$ of the
observable $Y_{i_k}$, and we write their sample mean as
\[
  \overline{Y}_k := \frac{1}{n_k}\sum_{\ell=1}^{n_k} Y_k^{(\ell)}.
\]
Samples at different scales are drawn independently (the project's
independence rule), so $\overline{Y}_1, \dots, \overline{Y}_K$ are independent
random variables.

\begin{assumption}[Relative variance, constant across scales]
\label{ass:relvar}
Write $\mu_k := \E\, Y_{i_k}$ for the (unknown) population mean at scale
$i_k$. We assume
\[
  \Var(Y_{i_k}) = \sigma^2 \mu_k^2
\]
for a single unknown constant $\sigma^2 > 0$, the same at every scale used.
This is the constant-variance regime of the article's Assumption~6
($\sigma_k^2 \to \sigma_\infty^2$; here we take $\sigma_k^2 \equiv \sigma^2$
exactly, consistent with $\xi_k := Y_{i_k}/\mu_k$ having $\Var(\xi_k) =
\sigma^2$ for every $k$ used).
\end{assumption}

\begin{assumption}[Leading-order power law]
\label{ass:powerlaw}
\[
  \mu_k = a_0\, i_k^{\gamma}
\]
for unknown constants $a_0 > 0$, $\gamma \in \mathbb{R}$ (the article's
eq.~232 with no finite-size correction term; see
Remark~\ref{rmk:corrections} for the extension).
\end{assumption}

\begin{assumption}[Gaussian approximation of the sample mean]
\label{ass:clt}
By the classical CLT (i.i.d., finite variance),
\[
  \sqrt{n_k}\,(\overline{Y}_k - \mu_k) \dto \mathcal{N}(0, \Var(Y_{i_k}))
  = \mathcal{N}(0, \sigma^2 \mu_k^2) \qquad \text{as } n_k \to \infty,
\]
and for the finite (but large) $n_k$ actually used we treat this as an
approximate equality in distribution:
\[
  \overline{Y}_k \ \approx\ \mathcal{N}\!\left(\mu_k,\ \frac{\sigma^2 \mu_k^2}{n_k}\right).
\]
\end{assumption}

The unknowns are $(\gamma, a_0, \sigma^2)$; the data are
$(i_k, n_k, \overline{Y}_k)_{k=1}^K$, all observed/known.

\section{The likelihood}

Under Assumption~\ref{ass:clt} and independence across $k$, the (approximate)
joint density of $(\overline{Y}_1,\dots,\overline{Y}_K)$ is
\[
  L(\gamma, a_0, \sigma^2)
  = \prod_{k=1}^K
    \left(2\pi\,\frac{\sigma^2\mu_k^2}{n_k}\right)^{-1/2}
    \exp\!\left[-\,\frac{n_k(\overline{Y}_k - \mu_k)^2}{2\sigma^2 \mu_k^2}\right],
  \qquad \mu_k = a_0 i_k^\gamma.
\]
Taking logarithms,
\begin{equation}
\label{eq:loglik}
  \ell(\gamma, a_0, \sigma^2)
  = -\frac{K}{2}\log(2\pi) - \frac{K}{2}\log\sigma^2
    - \sum_{k=1}^K \log \mu_k + \frac{1}{2}\sum_{k=1}^K \log n_k
    - \frac{1}{2\sigma^2}\sum_{k=1}^K \frac{n_k(\overline{Y}_k-\mu_k)^2}{\mu_k^2}.
\end{equation}

\section{Score equations}

We maximize \eqref{eq:loglik} over $(\gamma, a_0, \sigma^2)$ by setting its
partial derivatives to zero.

\subsection{With respect to $\sigma^2$}

\[
  \frac{\partial \ell}{\partial \sigma^2}
  = -\frac{K}{2\sigma^2} + \frac{1}{2\sigma^4}\sum_{k=1}^K \frac{n_k(\overline{Y}_k-\mu_k)^2}{\mu_k^2}
  = 0
\]
gives, in closed form,
\begin{equation}
\label{eq:sigmahat}
  \hat\sigma^2(\gamma, a_0) = \frac{1}{K}\sum_{k=1}^K \frac{n_k(\overline{Y}_k-\mu_k)^2}{\mu_k^2}.
\end{equation}

\subsection{With respect to $\mu_k$, then $\gamma$ and $a_0$}

Write $r_k := (\overline{Y}_k - \mu_k)/\mu_k$ for the relative residual at
scale $k$ (so $\overline{Y}_k = \mu_k(1+r_k)$). Differentiating the $k$-th
term of \eqref{eq:loglik} with respect to $\mu_k$ (holding $\sigma^2$ fixed),
using the quotient rule on $(\overline{Y}_k-\mu_k)^2/\mu_k^2$:
\[
  \frac{\partial}{\partial \mu_k}
  \left[\frac{(\overline{Y}_k-\mu_k)^2}{\mu_k^2}\right]
  = \frac{-2(\overline{Y}_k-\mu_k)\mu_k^2 - (\overline{Y}_k-\mu_k)^2 \cdot 2\mu_k}{\mu_k^4}
  = -\frac{2\overline{Y}_k(\overline{Y}_k-\mu_k)}{\mu_k^3},
\]
where the last equality uses $\mu_k + (\overline{Y}_k - \mu_k) = \overline{Y}_k$. Hence
\[
  \frac{\partial \ell}{\partial \mu_k}
  = -\frac{1}{\mu_k}
    + \frac{n_k\,\overline{Y}_k(\overline{Y}_k-\mu_k)}{\sigma^2 \mu_k^3}
  = \frac{1}{\mu_k}\left[\frac{n_k(1+r_k)r_k}{\sigma^2} - 1\right],
\]
using $\overline{Y}_k(\overline{Y}_k-\mu_k)/\mu_k^2 = (1+r_k)r_k$.

Since $\mu_k = a_0 i_k^\gamma$, we have
\[
  \frac{\partial \mu_k}{\partial \gamma} = \mu_k \log i_k,
  \qquad
  \frac{\partial \mu_k}{\partial a_0} = \frac{\mu_k}{a_0}.
\]
By the chain rule,
\begin{align}
  \frac{\partial \ell}{\partial \gamma}
  &= \sum_{k=1}^K \frac{\partial \ell}{\partial \mu_k}\cdot \mu_k \log i_k
   = \sum_{k=1}^K \log i_k
     \left[\frac{n_k(1+r_k)r_k}{\sigma^2} - 1\right] = 0,
   \label{eq:score-gamma}\\[4pt]
  \frac{\partial \ell}{\partial a_0}
  &= \frac{1}{a_0}\sum_{k=1}^K
     \left[\frac{n_k(1+r_k)r_k}{\sigma^2} - 1\right] = 0
   \ \Longleftrightarrow\
   \sum_{k=1}^K \left[\frac{n_k(1+r_k)r_k}{\sigma^2} - 1\right] = 0.
   \label{eq:score-a0}
\end{align}

\begin{remark}[A clean form for the $a_0$ equation]
Write $w_k := n_k r_k^2 = n_k(\overline{Y}_k-\mu_k)^2/\mu_k^2$, so that
\eqref{eq:sigmahat} reads $\sum_k w_k = K\hat\sigma^2$. Expanding
$n_k(1+r_k)r_k = n_k r_k + w_k$, equation~\eqref{eq:score-a0} becomes
$\sum_k (n_k r_k + w_k)/\sigma^2 = K$. At the joint stationary point
$\sigma^2=\hat\sigma^2$, so $\sum_k w_k/\hat\sigma^2 = K$ and this reduces to
\begin{equation}
\label{eq:score-a0-clean}
  \sum_{k=1}^K n_k\,\frac{\overline{Y}_k - \mu_k}{\mu_k} = 0,
\end{equation}
a weighted normal equation with no $\sigma^2$ dependence left --- useful as an
exact identity to check any numerical solution against. The $\gamma$
equation~\eqref{eq:score-gamma} does \emph{not} simplify the same way (the
extra $\log i_k$ weight blocks the same cancellation), so it must be kept in
its $\sigma^2$-dependent form, or solved jointly with
\eqref{eq:sigmahat}--\eqref{eq:score-a0-clean}.
\end{remark}

\section{Reduced form for implementation: the profile log-likelihood}

Equations \eqref{eq:sigmahat}, \eqref{eq:score-gamma}, \eqref{eq:score-a0} are
three equations in three unknowns with no closed-form solution for
$(\gamma, a_0)$ in general. The standard and most practical way to solve them
numerically is to \emph{profile out} $\sigma^2$: substitute
$\hat\sigma^2(\gamma,a_0)$ from \eqref{eq:sigmahat} back into
\eqref{eq:loglik}, which by construction removes $\sigma^2$ as a free
variable and leaves an objective in $(\gamma, a_0)$ alone.

Substituting \eqref{eq:sigmahat} into the last term of \eqref{eq:loglik}:
\[
  \frac{1}{2\hat\sigma^2}\sum_{k=1}^K \frac{n_k(\overline{Y}_k-\mu_k)^2}{\mu_k^2}
  = \frac{1}{2\hat\sigma^2}\cdot K\hat\sigma^2 = \frac{K}{2},
\]
so
\[
  \ell\big(\gamma, a_0, \hat\sigma^2(\gamma,a_0)\big)
  = -\frac{K}{2}\log(2\pi) - \frac{K}{2}\log\hat\sigma^2(\gamma,a_0)
    - \sum_{k=1}^K \log \mu_k + \frac{1}{2}\sum_{k=1}^K \log n_k - \frac{K}{2}.
\]
Dropping the additive constants that do not depend on $(\gamma, a_0)$,
maximizing the profile log-likelihood is equivalent to \emph{minimizing}
\begin{equation}
\label{eq:objective}
  \boxed{\
  \Phi(\gamma, a_0) \;:=\; K \log \hat\sigma^2(\gamma, a_0) \;+\; 2\sum_{k=1}^K \log \mu_k(\gamma, a_0),
  \qquad \mu_k = a_0 i_k^\gamma,\ \ \hat\sigma^2 \text{ as in \eqref{eq:sigmahat}}.
  \ }
\end{equation}
This is a plain function of two real variables $(\gamma, a_0)$.
\textbf{Caution:} it is tempting to hand $\Phi$ directly to a generic
numerical minimizer here, but Section~\ref{sec:second-order} shows this is
not reliable --- $\Phi$ is not established to be convex, and empirically is
often not. Do not skip ahead; use the algorithm in
Section~\ref{sec:implementation} instead.

\begin{remark}[Consistency check]
By the envelope theorem, $\partial \Phi/\partial\gamma = 0$ and
$\partial\Phi/\partial a_0 = 0$ are exactly equations \eqref{eq:score-gamma}
and \eqref{eq:score-a0} evaluated at $\sigma^2 = \hat\sigma^2(\gamma,a_0)$ ---
minimizing \eqref{eq:objective} and solving the original score equations are
the same problem. A numerical minimizer's solution should therefore also
satisfy the clean identity~\eqref{eq:score-a0-clean}; checking that is a
cheap correctness test on any implementation.
\end{remark}

\begin{remark}[Extension to the $J$-order expansion]
\label{rmk:corrections}
If instead $\mu_k = a_0 i_k^\gamma \exp\!\big(\sum_j a_j i_k^{-\omega_j}\big)$
(the article's full eq.~232, correction exponents $\omega_j$ treated as
known), every derivative above goes through unchanged: the correction factor
does not depend on $\gamma$ or $a_0$, so $\partial \mu_k/\partial\gamma =
\mu_k \log i_k$ and $\partial\mu_k/\partial a_0 = \mu_k/a_0$ still hold
exactly. Equations \eqref{eq:score-gamma}--\eqref{eq:objective} are therefore
unchanged in form; only the definition of $\mu_k$ changes.
\end{remark}

\begin{remark}[How this differs from the OLS-on-log estimator]
\label{rmk:vs-ols}
\texttt{tools/loglog.py}'s existing estimators work by treating
$\log\overline{Y}_k$ as approximately Gaussian around $\log\mu_k$ (a
delta-method consequence of Assumption~\ref{ass:clt}, valid when the relative
fluctuation $\sigma/\sqrt{n_k}$ is small) and fitting an \emph{unweighted}
OLS line to $(\log i_k, \log \overline{Y}_k)$. The estimator derived here
instead applies the CLT directly to $\overline{Y}_k$, without a log
transform, and consequently weighs each scale by $n_k/\mu_k^2$ (visible in
\eqref{eq:score-a0-clean} and in $\hat\sigma^2$) rather than uniformly. The
two coincide asymptotically as $n_k \to \infty$ (delta method), but are
different estimators at finite $n_k$, and comparing them empirically was the
point of building \texttt{compare\_methods} in the first place.
\end{remark}

\section{Second-order conditions: is this actually a maximum?}
\label{sec:second-order}

Setting the score equations to zero, as done above, only identifies
\emph{stationary} points of $\ell$ --- it says nothing about whether a given
stationary point is a maximum, a saddle, or (in the profiled objective
$\Phi$, which is \emph{minimized}) a minimum. This was not checked in the
first version of this note. We check it now, and the answer is: yes for part
of the problem, provably; no in general for the rest, and this matters enough
to change the recommended algorithm.

Throughout this section write $\beta := \log a_0 \in \mathbb{R}$ (so
$a_0 = e^\beta$, removing the $a_0>0$ constraint) and
$\eta_k := \log \mu_k = \gamma \log i_k + \beta$, which is \emph{affine} in
$(\gamma,\beta)$ --- unlike $\mu_k = a_0 i_k^\gamma$, which is affine in
neither $(\gamma,a_0)$ nor $(\gamma,\beta)$. This reparametrization is a
smooth, monotonic bijection in $a_0$, so it does not change which $a_0$ is
optimal; it only makes the curvature analysis below exact instead of
approximate.

\subsection{The likelihood is concave in \texorpdfstring{$(\gamma,\beta)$}{(gamma,beta)}, at fixed \texorpdfstring{$\sigma^2$}{sigma^2}, on an explicit region}

Write $x_k := \overline{Y}_k/\mu_k = \overline{Y}_k e^{-\eta_k}$ (so $x_k=1+r_k$
in the earlier notation). Differentiating the $k$-th term of
\eqref{eq:loglik} twice with respect to $\eta_k$ at fixed $\sigma^2$ gives
\begin{equation}
\label{eq:ellkpp}
  \frac{\partial^2 \ell_k}{\partial \eta_k^2}
  = -\frac{n_k\, x_k (2x_k - 1)}{\sigma^2}
\end{equation}
(verified symbolically). Since $x_k>0$ always, this is $\le 0$ exactly when
$x_k \ge \tfrac12$, i.e.\ when
\begin{equation}
\label{eq:goodregion}
  \mu_k < 2\,\overline{Y}_k.
\end{equation}
This is a mild condition: it fails only if the model mean overshoots the
observed scale mean by more than a factor of two, which should not happen
anywhere near a reasonable fit.

Because $\eta_k(\gamma,\beta) = (\log i_k)\,\gamma + \beta$ is affine, writing
$v_k := (\log i_k,\ 1)^\top \in \mathbb{R}^2$, the chain rule gives the
Hessian of $\ell(\gamma,\beta;\sigma^2) = \sum_k \ell_k(\eta_k(\gamma,\beta))$
directly as a sum of rank-one terms:
\begin{equation}
\label{eq:hessian-outer}
  \nabla^2_{(\gamma,\beta)}\, \ell
  = \sum_{k=1}^K \frac{\partial^2 \ell_k}{\partial \eta_k^2}\, v_k v_k^\top
  = -\sum_{k=1}^K w_k\, v_k v_k^\top,
  \qquad w_k := \frac{n_k x_k(2x_k-1)}{\sigma^2}.
\end{equation}
For any $u \in \mathbb{R}^2$, $u^\top \nabla^2\ell\, u = -\sum_k w_k (v_k\cdot u)^2$.
If $w_k \ge 0$ for every $k$ (i.e.\ \eqref{eq:goodregion} holds for every
$k$), every term is $\le 0$, so $\nabla^2 \ell \preceq 0$: \textbf{$\ell$ is
concave in $(\gamma,\beta)$, at fixed $\sigma^2$, on the region}
$R := \{(\gamma,\beta) : \mu_k(\gamma,\beta) < 2\overline{Y}_k \ \forall k\}$,
an intersection of $K$ half-planes, hence convex. Moreover $u^\top\nabla^2\ell\, u = 0$
forces $u \perp v_k$ for every $k$ with $w_k>0$; since the scales
$i_1<\cdots<i_K$ are pairwise distinct, any two $v_k$'s are linearly
independent, so as soon as \emph{two} scales satisfy \eqref{eq:goodregion}
strictly, $u=0$ is forced and $\ell$ is \emph{strictly} concave on $R$ --- a
unique maximizer within $R$, given $\sigma^2$.

This was checked both symbolically (verifying \eqref{eq:ellkpp} with
\texttt{sympy}) and numerically: on five independent simulated realizations
of a $K=4$ synthetic experiment ($i_k=2^{10},2^{13},2^{16},2^{19}$,
$n_k\equiv5000$, true $\gamma=0.5,a_0=1,\sigma^2=0.04$), the Hessian
\eqref{eq:hessian-outer} evaluated at the true parameters was positive
definite for $-\ell$ (i.e.\ $\ell$ concave) in all five trials, matching
\eqref{eq:goodregion} holding at every scale in every trial.

\subsection{The profiled objective \texorpdfstring{$\Phi$}{Phi} does \emph{not} inherit this}

Concavity of $\ell(\gamma,\beta;\sigma^2)$ at a \emph{fixed} $\sigma^2$ does
not by itself imply anything about $\Phi(\gamma,\beta) = \ell(\gamma,\beta,\hat\sigma^2(\gamma,\beta))$,
where $\sigma^2$ is also allowed to vary with $(\gamma,\beta)$: profiling out
a parameter preserves concavity only when the \emph{joint} log-likelihood is
concave in (here) $(\eta,\sigma^2)$ together, which is a strictly stronger
condition that the $-\frac{K}{2}\log\sigma^2$ term and its coupling to the
residuals through $\eta$ does not obviously satisfy, and which we did not
prove.

We checked directly instead. Using finite differences on the same five
simulated realizations as above, the numerical Hessian of $\Phi$ at the
\emph{true} parameters had a negative eigenvalue in three of the five trials
--- in one case, a large negative eigenvalue exceeding the positive one in
magnitude, i.e.\ $\Phi$ was locally concave, not convex, in a direction right
at the true parameter values. \textbf{$\Phi$ should not be treated as
reliably convex}, even close to the truth, and handing it directly to a
generic minimizer (as the previous version of this note recommended) is not
trustworthy: the minimizer can converge to a spurious stationary point
without any warning.

\subsection{Fix: coordinate ascent, using only the part that is provably well-posed}

The one part of this problem that \emph{is} provably well-behaved is
maximizing $\ell$ over $(\gamma,\beta)$ at a \emph{fixed} $\sigma^2$
(Section~\ref{sec:second-order}.1). This suggests block coordinate ascent on
the joint log-likelihood $\ell(\gamma,\beta,\sigma^2)$, alternating exact
maximization over each block:
\begin{enumerate}
\item[(a)] fix $\sigma^2$; maximize $\ell(\gamma,\beta;\sigma^2)$ over
      $(\gamma,\beta)$ --- concave, uniquely solvable on $R$ (e.g.\ Newton's
      method, which converges fast on a genuinely concave problem);
\item[(b)] fix $(\gamma,\beta)$; update $\sigma^2$ in closed form via
      \eqref{eq:sigmahat}.
\end{enumerate}
Each step exactly maximizes over its own block holding the other fixed, so
$\ell$ is non-decreasing after every step (a standard property of exact
coordinate ascent) --- this converges to \emph{some} stationary point, though
not provably the global maximum, since $\ell$ is not established to be
jointly concave overall. This is a materially weaker but honest claim, and a
far more trustworthy algorithm than minimizing $\Phi$ directly, since every
individual step is exactly the sub-problem we proved is well-posed.

This was implemented and tested (200 replicates of the same $K=4$ synthetic
setup): the iteration converges in a handful of steps per replicate ($\le 3$
in every case tried), and recovered $\gamma$ with bias and RMSE
\emph{statistically indistinguishable} from the existing OLS-on-log
estimator in this particular design. That is expected, not a null result: with
$n_k$ equal across scales, the weights $w_k = n_k x_k(2x_k-1)/\sigma^2$ in
\eqref{eq:hessian-outer} become (approximately) equal across $k$ near the
optimum ($x_k\approx1$), so the weighted problem coincides with the
unweighted one in this symmetric case. The two estimators should differ when
$n_k$ varies across scales (unequal sampling effort per scale is exactly the
budget-allocation regime the article's own Theorem~CLT and optimal-allocation
results are about) --- an empirical comparison worth running once this is
implemented, via \texttt{compare\_methods}.

\section{Reconsidering: is coordinate ascent actually necessary?}
\label{sec:direct}

Coordinate ascent (Section~\ref{sec:second-order}.3) was motivated by
$\Phi$'s convexity being unproven and empirically often false. But observe:
if the \emph{joint} log-likelihood $\ell(\gamma,\beta,\sigma^2)$ were jointly
concave, its profile $\Phi$ would necessarily be concave too (partial
maximization preserves concavity). Contrapositive: since $\Phi$ is not
concave, $\ell$ is \emph{not} jointly concave either. So directly maximizing
$\ell$ over all three parameters at once carries exactly the same
\emph{unproven global optimality} as minimizing $\Phi$ directly --- eliminating
$\sigma^2$ algebraically first, versus leaving it in and optimizing jointly,
does not change the shape of the underlying surface. What coordinate ascent
actually bought was narrower than it looked: only step~(a) of each iteration
is individually provably well-posed; the overall iteration's convergence to
the \emph{global} optimum was never guaranteed either.

Given that, the two approaches' honest guarantees are closer than they first
appear, and it is worth checking whether the simpler one --- a single generic
optimizer call over $(\gamma,\beta,\log\sigma^2)$ jointly, from the
OLS-on-log starting point --- behaves comparably in practice, carrying the
\emph{same} post-hoc diagnostics either way (since neither algorithm's output
should be trusted without them).

Checked on the same 200-replicate, $K=4$ setup, with \texttt{scipy}'s
L-BFGS-B: the recovered $\hat\gamma$ matched coordinate ascent's bias and
RMSE to six decimal places. The optimizer's own convergence flag reported
failure in 5 of 200 runs (2.5\%) --- in every one of those, $\hat\gamma$ was
still close to the truth, and both \eqref{eq:goodregion} and the (now full
three-parameter) Hessian check still passed; only the optimizer's internal
stopping criterion was unsatisfied. So: simpler, and empirically no worse
here, \emph{provided} the diagnostics are checked and any non-convergence is
surfaced rather than silently ignored --- which is what
Section~\ref{sec:implementation} now does, in place of coordinate ascent.
Coordinate ascent remains available (Section~\ref{sec:second-order}.3) as a
more conservative fallback if direct optimization's diagnostics ever fail in
practice on real (non-synthetic) data.

\section{Summary for implementation}
\label{sec:implementation}

Given $(i_k, n_k, \overline{Y}_k)_{k=1}^K$ (implemented in
\texttt{tools/loglog.py}'s \texttt{gamma\_mle}):
\begin{enumerate}
\item Reparametrize $a_0 = e^\beta$, $\sigma^2 = e^{\log\sigma^2}$ (removes
      both positivity constraints); initialize $(\gamma,\beta)$ from the
      existing OLS-on-log estimate (\texttt{gamma\_all\_points}), and
      $\log\sigma^2 = 0$.
\item Minimize $-\ell(\gamma,\beta,\log\sigma^2)$ of \eqref{eq:loglik} directly
      over all three parameters jointly (\texttt{scipy.optimize.minimize},
      L-BFGS-B).
\item Report $\hat\gamma$, $\hat a_0=e^{\hat\beta}$,
      $\hat\sigma^2=e^{\widehat{\log\sigma^2}}$, \textbf{together with}:
      \begin{itemize}
      \item \texttt{converged}: the optimizer's own success flag;
      \item \texttt{region\_ok}: \eqref{eq:goodregion},
            $\hat\mu_k < 2\overline{Y}_k$ for every $k$;
      \item \texttt{hessian\_pd}: the numerical Hessian of $-\ell$ at the
            solution, over all three parameters, is positive definite
            (confirms a genuine local maximum of $\ell$, not a saddle);
      \item \texttt{trustworthy} := all three of the above.
      \end{itemize}
\end{enumerate}
\textbf{\texttt{gamma\_hat} should not be used without checking
\texttt{trustworthy}} --- this is not decorative, per
Section~\ref{sec:direct}. Validated (200 synthetic replicates,
$i_k=2^{10},2^{13},2^{16},2^{19}$, $n_k\equiv5000$, true
$\gamma=0.5,a_0=1,\sigma^2=0.04$): \texttt{trustworthy} was \texttt{False} in
exactly the same 5/200 trials found in Section~\ref{sec:direct} (via
